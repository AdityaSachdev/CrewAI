# Calculate the first 10,000 terms of the series 1/1 - 1/3 + 1/5 - 1/7 + ... and multiply the sum by 4

num_terms = 10000

# Initialize sum
series_sum = 0.0

for i in range(num_terms):
    # Calculate term
    term = (-1)**i / (2*i + 1)
    series_sum += term

# Multiply by 4
result = 4 * series_sum

print(result)
